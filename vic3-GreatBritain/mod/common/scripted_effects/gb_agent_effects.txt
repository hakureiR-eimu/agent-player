# 游戏接入 mod 侧：年度状态导出逻辑
# 每年（游戏内）把世界格局写进 debug.log：rank 1-16 每国一行国家行，GBR 追加一行财政行。
# 行格式是 mod 侧与 mcp 侧的交接面，见 docs/prd/游戏接入-mod侧.md；字段取值写法的可行性随首次跑 mod 实测，
# 回退路径写在本文件各字段注释与 prd 待实测清单里。
#
# 内插语法说明：debug_log 的消息支持 data function 内插，前缀 [THIS.GetCountry.XXX]
# 是原版脚本的标准写法（00_code_on_actions.txt 等多处 debug_log 用例）。

gb_agent_export_state = {
	# 国家行：列强＋次强（rank 1-16），每国一项 GDP、一项常备军
	if = {
		limit = {
			# 比较国家世界名次的 trigger（wiki Trigger 文档已证，同族示例 global_country_ranking > 42）
			global_country_ranking <= 16
		}
		# 各字段（均 wiki GUI 脚本函数表已证存在于 Country 对象）：
		#   tag   GetTagName      输出是否为三字码待实测
		#   year  TimeKeeper.GetCurrentDate.GetString（原版 debug_log 用例写法），
		#         输出为完整日期字符串，是否纯年份待实测；不行为纯年时试 Date.Format 或 mcp 侧截取
		#   rank  GetRankNumber   国家名次数字，是否即 world ranking 待实测
		#   gdp   GetGDP          与 trigger gdp 同义（yearly predicted GDP）
		#   army  GetNumBattalions 是否排除已征召兵待实测；不符则换 GetBattalions
		debug_log = "vic3-GreatBritain state tag=[THIS.GetCountry.GetTagName] year=[TimeKeeper.GetCurrentDate.GetString] rank=[THIS.GetCountry.GetRankNumber] gdp=[THIS.GetCountry.GetGDP] army=[THIS.GetCountry.GetNumBattalions]"
	}

	# 财政行：仅大不列颠追加；不受上面 16 强门槛约束（GBR 跌出名次财政也要记）
	if = {
		limit = {
			# scope 相等比较的原版写法（00_oregon.txt：c:USA = ROOT）
			c:GBR = ROOT
		}
		# gold     GetGoldReserves   金库（wiki Trigger 文档 gold_reserves 同义）
		# debt     GetPrincipal      债务本金（wiki Trigger 文档 principal 同义）
		# week_net GetWeeklyBalance  每周净收支；与 trigger net_total_income（收抵支后的余量）的
		#          对应关系待实测，不符则回退记周收入 GetWeeklyIncome（trigger income 同义）
		debug_log = "vic3-GreatBritain fin tag=[THIS.GetCountry.GetTagName] year=[TimeKeeper.GetCurrentDate.GetString] gold=[THIS.GetCountry.GetGoldReserves] debt=[THIS.GetCountry.GetPrincipal] week_net=[THIS.GetCountry.GetWeeklyBalance]"
	}
}
