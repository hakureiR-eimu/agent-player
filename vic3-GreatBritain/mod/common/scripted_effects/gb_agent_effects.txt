# 游戏接入 mod 侧：年度状态导出逻辑
# 每年（游戏内）把世界格局写进 debug.log：rank 1-16 每国一行国家行，GBR 追加一行财政行。
# 行格式是 mod 侧与 mcp 侧的交接面，见 docs/prd/游戏接入-mod侧.md；字段取值写法的可行性随首次跑 mod 实测，
# 回退路径写在本文件各字段注释与 prd 待实测清单里。
#
# 内插语法说明：debug_log 的消息支持 data function 内插，前缀 [THIS.GetCountry.XXX]
# 是原版脚本的标准写法（00_code_on_actions.txt 等多处 debug_log 用例）。

gb_agent_export_state = {
	# 国家行：列强＋次强（rank 1-16），每国一项 GDP、一项常备军
	if = {
		limit = {
			# 比较国家世界名次的 trigger（wiki Trigger 文档已证，同族示例 global_country_ranking > 42）
			global_country_ranking <= 16
		}
		# 各字段（均 wiki GUI 脚本函数表已证存在于 Country 对象；2026-11-20 首次实测收敛见 prd 待实测清单）：
		#   tag   GetTagName      已实测：输出三字码
		#   year  TimeKeeper.GetCurrentDate.GetString，已实测输出完整本地化日期（如"6月 13, 1836"，
		#         含空格逗号），year 定案由 mcp 侧从值尾抽 4 位年份数字；行内含空格，mcp 解析须按字段名取值不按空格切分
		#   rank  GetCountryScorePosition 世界名次，已实测（1837 年 16 行取值 1-16，GBR=1）；
		#         首版用过的 GetRankNumber 实测输出的是国家分级（5-7）非名次，勿回退
		#   gdp   GetGDP          已实测：原始小数无千分位无缩写
		#   army  GetNumBattalions 营队数（开局值合理；征召期口径遇战争年顺带看）
		debug_log = "vic3-GreatBritain state tag=[THIS.GetCountry.GetTagName] year=[TimeKeeper.GetCurrentDate.GetString] rank=[THIS.GetCountry.GetCountryScorePosition] gdp=[THIS.GetCountry.GetGDP] army=[THIS.GetCountry.GetNumBattalions]"
	}

	# 财政行：仅大不列颠追加；不受上面 16 强门槛约束（GBR 跌出名次财政也要记）
	if = {
		limit = {
			# scope 相等比较的原版写法（00_oregon.txt：c:USA = ROOT）
			c:GBR = ROOT
		}
		# gold     GetGoldReserves   金库（wiki Trigger 文档 gold_reserves 同义）；已实测输出原始小数
		# debt     GetPrincipal      债务本金（wiki Trigger 文档 principal 同义）；已实测输出原始小数
		# week_net GetWeeklyBalance  每周净收支；首次实测输出带符号原始小数（开局 GBR -10.8万/周），
		#          量级合理，与 trigger net_total_income 的严格同值无从交叉验证，按界面语义采信
		debug_log = "vic3-GreatBritain fin tag=[THIS.GetCountry.GetTagName] year=[TimeKeeper.GetCurrentDate.GetString] gold=[THIS.GetCountry.GetGoldReserves] debt=[THIS.GetCountry.GetPrincipal] week_net=[THIS.GetCountry.GetWeeklyBalance]"
	}
}
